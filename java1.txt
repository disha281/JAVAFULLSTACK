class Solution {
    public String formatIdCard(String name, String usn, int semester) {

        if (semester < 1 || semester > 8) {
            return "INVALID";
        }

        name = name.trim();
        usn = usn.trim().toUpperCase();

        String[] words = name.split("\\s+");
        StringBuilder formattedName = new StringBuilder();

        for (String word : words) {
            formattedName.append(
                word.substring(0, 1).toUpperCase()
            );

            if (word.length() > 1) {
                formattedName.append(
                    word.substring(1).toLowerCase()
                );
            }

            formattedName.append(" ");
        }

        return formattedName.toString().trim()
                + " | "
                + usn
                + " | S"
                + semester;
    }
}

  class Solution {
    public int countDistinctPayments(String[] receipts) {

        HashSet<String> set = new HashSet<>();

        for (String receipt : receipts) {

            String[] parts = receipt.split("\\|");

            String usn = parts[0].toLowerCase();

            long amount = Long.parseLong(parts[1]);

            String key = usn + "|" + amount;

            set.add(key);
        }

        return set.size();
    }
}

interface Payable {
    long amountPaise();
}

class Payment implements Payable {
    private final long paise;
    Payment(long paise) { this.paise = paise; }
    public long amountPaise() { return paise; }
}

class LateFeePayment implements Payable {
    private final Payment base;
    private final long finePaise;

    LateFeePayment(Payment base, long finePaise) {
        this.base = base;
        this.finePaise = finePaise;
    }

    public long amountPaise() {
        return base.amountPaise();   // the fine is missing
    }
}

import java.util.*;

class Solution {
    public String applyRoomOps(String[] ops) {
        Map<String, String> studentRoom = new HashMap<>();
        Map<String, Integer> roomCount = new HashMap<>();

        int applied = 0;
        int rejected = 0;

        for (String op : ops) {

            if (op.startsWith("ASSIGN:")) {
                String[] parts = op.split(":");
                String student = parts[1];
                String room = parts[2];

                // Rule 1: student already has a room
                if (studentRoom.containsKey(student)) {
                    rejected++;
                    continue;
                }

                // Rule 2: room already has 2 residents
                int count = roomCount.getOrDefault(room, 0);

                if (count >= 2) {
                    rejected++;
                    continue;
                }

                // Apply assignment
                studentRoom.put(student, room);
                roomCount.put(room, count + 1);

                applied++;

            } else { // VACATE
                String student = op.substring(7);

                // Student has no room
                if (!studentRoom.containsKey(student)) {
                    rejected++;
                    continue;
                }

                // Find the room the student occupies
                String room = studentRoom.get(student);

                // Decrease room count
                int count = roomCount.get(room);
                roomCount.put(room, count - 1);

                // Remove student's assignment
                studentRoom.remove(student);

                applied++;
            }
        }

        return applied + "," + rejected;
    }
}
